    70→    70→function splitSentences(text: string): string[] {
    71→    71→  const raw: string[] = [];
    72→    72→  let cur = "";
    73→    73→  for (const ch of text) {
    74→    74→    if (".!?؟۔".includes(ch) || ch === "\n") {
    75→    75→      const t = cur.trim();
    76→    76→      if (t) raw.push(t);
    77→    77→      cur = "";
    78→    78→    } else {
    79→    79→      cur += ch;
    80→    80→    }
    81→    81→  }
    82→    82→  if (cur.trim()) raw.push(cur.trim());
    83→    83→
    84→    84→  const merged: string[] = [];
    85→    85→  for (const part of raw) {
    86→    86→    const last = merged[merged.length - 1];
    87→    87→    if (last && (last.length < 14 || part.length < 14) && `${last} ${part}`.length <= 300) {
    88→    88→      merged[merged.length - 1] = `${last} ${part}`;
    89→    89→    } else {
    90→    90→      merged.push(part.slice(0, 320));
    91→    91→    }
    92→    92→  }
    93→    93→  return merged.slice(0, 6);
    94→    94→}

    96→    96→function parseReply(reply: string): { items: QueueItem[]; plain: string } {
    97→    97→  const match = reply.match(EMOTION_RE);
    98→    98→  const emotion = (match && EMO_LIST.includes(match[1] as Emotion) ? match[1] : "neutral") as Emotion;
    99→    99→  const plain = reply.replace(EMOTION_RE, "").trim();
   100→   100→  const items = splitSentences(plain).map((text) => ({ text, emotion }));
   101→   101→  return { items, plain };
   102→   102→}

   190→   190→async function fetchBlob(text: string, lang: Lang, emotion: Emotion): Promise<Blob> {
   191→   191→  const ctrl = new AbortController();
   192→   192→  const timer = setTimeout(() => ctrl.abort(), 20000);
   193→   193→  try {
   194→   194→    const res = await fetch("/api/voice-agent/tts", {
   195→   195→      method: "POST",
   196→   196→      headers: { "Content-Type": "application/json" },
   197→   197→      body: JSON.stringify({ text, lang, emotion }),
   198→   198→      signal: ctrl.signal,
   199→   199→    });
   200→   200→    if (!res.ok) throw new Error(`tts ${res.status}`);
   201→   201→    const blob = await res.blob();
   202→   202→    if (blob.size < 500) throw new Error("tiny audio");
   203→   203→    return blob;
   204→   204→  } finally {
   205→   205→    clearTimeout(timer);
   206→   206→  }
   207→   207→}
