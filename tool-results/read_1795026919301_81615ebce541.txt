    70→function splitSentences(text: string): string[] {
    71→  const raw: string[] = [];
    72→  let cur = "";
    73→  for (const ch of text) {
    74→    if (".!?؟۔".includes(ch) || ch === "\n") {
    75→      const t = cur.trim();
    76→      if (t) raw.push(t);
    77→      cur = "";
    78→    } else {
    79→      cur += ch;
    80→    }
    81→  }
    82→  if (cur.trim()) raw.push(cur.trim());
    83→
    84→  const merged: string[] = [];
    85→  for (const part of raw) {
    86→    const last = merged[merged.length - 1];
    87→    if (last && (last.length < 14 || part.length < 14) && `${last} ${part}`.length <= 300) {
    88→      merged[merged.length - 1] = `${last} ${part}`;
    89→    } else {
    90→      merged.push(part.slice(0, 320));
    91→    }
    92→  }
    93→  return merged.slice(0, 6);
    94→}

    96→function parseReply(reply: string): { items: QueueItem[]; plain: string } {
    97→  const match = reply.match(EMOTION_RE);
    98→  const emotion = (match && EMO_LIST.includes(match[1] as Emotion) ? match[1] : "neutral") as Emotion;
    99→  const plain = reply.replace(EMOTION_RE, "").trim();
   100→  const items = splitSentences(plain).map((text) => ({ text, emotion }));
   101→  return { items, plain };
   102→}

   190→async function fetchBlob(text: string, lang: Lang, emotion: Emotion): Promise<Blob> {
   191→  const ctrl = new AbortController();
   192→  const timer = setTimeout(() => ctrl.abort(), 20000);
   193→  try {
   194→    const res = await fetch("/api/voice-agent/tts", {
   195→      method: "POST",
   196→      headers: { "Content-Type": "application/json" },
   197→      body: JSON.stringify({ text, lang, emotion }),
   198→      signal: ctrl.signal,
   199→    });
   200→    if (!res.ok) throw new Error(`tts ${res.status}`);
   201→    const blob = await res.blob();
   202→    if (blob.size < 500) throw new Error("tiny audio");
   203→    return blob;
   204→  } finally {
   205→    clearTimeout(timer);
   206→  }
   207→}

   407→            }

   410→          u.onerror = finish;

   413→        } catch {
   414→          resolve();
   415→        }
   416→      });

   436→        }
   437→      };

   450→        } else {
   451→          await speakFallback(jobs[i].item.text, lang, jobs[i].item.emotion, runId);
   452→        }
   453→        if (runId !== runIdRef.current) return;
   454→      }
   455→
   456→      if (runId === runIdRef.current) {
   457→        setState(elConvRef.current ? "listening" : "idle");
   458→        setCaption("");
   459→      }
   460→    },
   461→    [playBlob, speakFallback],

   464→  const speakReply = useCallback(
   465→    (reply: string) => {

   471→        { role: "assistant" as const, content: plain },

   476→        setState("idle");
   477→      }
   478→    },
   479→    [runQueue],

   482→  const pushUserTurn = useCallback((text: string) => {

   498→          headers: { "Content-Type": "application/json" },
   499→          body: JSON.stringify({ messages: historyRef.current }),

   501→        const data = (await res.json()) as { reply?: string; error?: string; booking?: boolean };
   502→        if (!res.ok || !data.reply) throw new Error(data.error ?? "brain failed");
   503→        if (data.booking) setBookHint(true);
   504→        speakReply(data.reply);
   505→      } catch {
   506→        setMicHint("");
   507→        speakReply(localBrain(text));
   508→      } finally {
   509→        thinkingRef.current = false;
   510→      }
   511→    },

   521→      setCaption("");

   529→          elConvRef.current.sendUserMessage(text);
   530→          return;
   531→        } catch {
   532→          await endElSession();
   533→        }
   534→      }
   535→
   536→      // 2) try starting a premium session (voice first; legacy on failure)
   537→      if (!premiumFailedRef.current) {
   538→        setState("thinking");
   539→        const ok = await startElVoice();
   540→        if (ok && elConvRef.current) {
   541→          // the session plays the agent's Urdu greeting; forward the question after it
   542→          setTimeout(() => {
   543→            try {
   544→              elConvRef.current?.sendUserMessage(text);
   545→            } catch {}
   546→          }, 2200);
   547→          return;
   548→        }
   549→        await endElSession();
   550→      }
   551→
   552→      // 3) legacy pipeline (pushes its own user turn)
   553→      setMicHint(premiumFailedRef.current ? EL_FAILED_HINT : "");
   554→      void legacySend(text);
   555→    },
   556→    [endElSession, legacySend, pushUserTurn, startElVoice, stopAll],
   557→  );

   559→  /* ---------- mic + visualizer (legacy listening path) ---------- */

   561→  const startVisual = useCallback(async () => {

   566→      ctxRef.current = ctx;
   567→      const analyser = ctx.createAnalyser();

   569→      ctx.createMediaStreamSource(stream).connect(analyser);
   570→      const data = new Uint8Array(analyser.frequencyBinCount);
   571→      const loop = () => {
   572→        if (!ctxRef.current) return;
   573→        analyser.getByteFrequencyData(data);
   574→        barsRef.current.forEach((b, i) => {
   575→          if (!b) return;
   576→          const v = data[3 + i * 5] / 255;
   577→          b.style.transform = `scaleY(${(0.22 + v * 1.6).toFixed(2)})`;
   578→        });
   579→        rafRef.current = requestAnimationFrame(loop);
   580→      };
   581→      loop();
   582→    } catch {
   583→      barsRef.current.forEach((b, i) => {
   584→        if (b) b.classList.add("va-bar-fallback");
   585→      });
   586→    }
   587→  }, []);
   588→
   589→  const stopListening = useCallback(() => {
   590→    try {
   591→      recRef.current?.stop();
   592→    } catch {}
   593→    stopVisual();
   594→    if (stateRef.current === "listening") setState("idle");
   595→  }, [stopVisual]);
   596→
   597→  const startListening = useCallback(() => {
   598→    const Ctor = (
   599→      window as unknown as {
   600→        SpeechRecognition?: SRCtor;
   601→        webkitSpeechRecognition?: SRCtor;
   602→      }
   603→    ).SpeechRecognition ||
   604→    (window as unknown as { webkitSpeechRecognition?: SRCtor }).webkitSpeechRecognition;
   605→
   606→    if (premiumFailedRef.current || !Ctor) {
   607→      if (!Ctor) {
   608→        setVoiceInputOk(false);
   609→        setMicHint("Is browser mein voice input support nahi — text se poochein.");
   610→      } else {
   611→        setMicHint(EL_FAILED_HINT);
   612→      }
   613→      return;
   614→    }
   615→
   616→    stopAll();
   617→    setCaption("");
   618→    setInterim("");
   619→    setMicHint("");
   620→    setState("listening");
   621→
   622→    void (async () => {
   623→      const ok = await startElVoice();
   624→      if (!ok) {
   625→        // mic may be denied or quota hit -> fall back to browser recognition + legacy pipeline
   626→        stopVisual();
   627→        setState("idle");
   628→        if (premiumFailedRef.current) {
   629→          setMicHint(EL_FAILED_HINT);
   630→        } else {
   631→          setVoiceInputOk(false);
   632→          setMicHint(MIC_DENIED);
   633→        }
   634→      }
   635→    })();
   636→  }, [startElVoice, stopAll, stopVisual]);
   637→
   638→  const endVoiceSession = useCallback(() => {
   639→    stopAll();
   640→    stopVisual();
   641→    void endElSession();
   642→    setState("idle");
   643→    setCaption("");
   644→  }, [endElSession, stopAll, stopVisual]);
   645→
   646→  /* ---------- greeting + cleanup ---------- */
   647→
   648→  const openPanel = useCallback(() => {
   649→    setOpen(true);
   650→  }, []);
   651→
   652→  const closePanel = useCallback(() => {
   653→    stopAll();
   654→    stopListening();
   655→    void endElSession();
   656→    setOpen(false);
   657→  }, [endElSession, stopAll, stopListening]);
   658→
   659→  useEffect(() => {
   660→    return () => {
   661→      stopAll();
   662→      stopVisual();
   663→      void endElSession();
   664→    };
   665→  }, [endElSession, stopAll, stopVisual]);
   666→
   667→  /* ---------- render ---------- */
   668→
   669→  const statusText =
   670→    state === "listening"
   671→      ? premiumMode === "voice"
   672→        ? "Sun rahi hoon — boliye!"
   673→        : "Sun rahi hoon..."
   674→      : state === "thinking"
   675→        ? "Soch rahi hoon..."
   676→        : state === "speaking"
   677→          ? "Jawab de rahi hoon — boliye to beech mein rok doongi"
   678→          : "Boliye ya likhiye — jawab awaz mein milega";
   679→
   680→  const orbStyle = { bottom: "calc(1.15rem + env(safe-area-inset-bottom))", right: "1.15rem" };
   681→
   682→  if (!open) {
   683→    return (
   684→      <button
   685→        type="button"
   686→        onClick={openPanel}
   687→        aria-label="TK Dent voice assistant kholen"
   688→        className="va-orb fixed z-[80] flex h-14 w-14 items-center justify-center rounded-full bg-gradient-to-br from-teal to-[#07272B] text-white shadow-xl transition-transform duration-300 hover:scale-110 focus-visible:outline focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-teal sm:h-16 sm:w-16"
   689→        style={orbStyle}
   690→      >
   691→        <AudioLines className="h-6 w-6 sm:h-7 sm:w-7" aria-hidden="true" />
   692→      </button>
   693→    );
   694→  }
   695→
   696→  return (
   697→    <section
   698→      role="dialog"
   699→      aria-label="TK Dent voice assistant Sana"
   700→      className="fixed z-[80] w-[min(92vw,380px)] overflow-hidden rounded-3xl border border-teal/25 bg-white/90 shadow-2xl backdrop-blur-xl"
   701→      style={orbStyle}
   702→    >
   703→      {/* header */}
   704→      <header className="flex items-center gap-3 border-b border-teal/15 bg-gradient-to-r from-teal/10 to-mint/20 px-4 py-3">
   705→        <span className="flex h-9 w-9 items-center justify-center rounded-full bg-gradient-to-br from-teal to-[#07272B] text-white">
   706→          <AudioLines className="h-4 w-4" aria-hidden="true" />
   707→        </span>
   708→        <span className="min-w-0 flex-1">
   709→          <span className="block text-sm font-bold leading-tight text-ink">Sana</span>
   710→          <span className="block truncate text-[11px] leading-tight text-muted-ink">
   711→            {premiumMode === "voice" ? "TK Dent Voice Assistant · Premium" : "TK Dent Voice Assistant"}
   712→          </span>
   713→        </span>
   714→        <div
   715→          className="flex items-center gap-1 rounded-full border border-teal/25 bg-white/70 p-0.5"
   716→          role="group"
   717→          aria-label="Boli jani wali zuban"
   718→        >
   719→          {(["ur", "en"] as Lang[]).map((l) => (
   720→            <button
   721→              key={l}
   722→              type="button"
   723→              onClick={() => setInputLang(l)}
   724→              aria-pressed={inputLang === l}

   828→        >

   830→          WhatsApp par appointment book karein

   864→              setDraft("");

   867→          placeholder={state === "speaking" ? "Sana bol rahi hai..." : "Apna sawal likhein..."}

   871→        />

   873→        {state === "speaking" && premiumMode !== "voice" ? (

   895→            disabled={!draft.trim() || state === "thinking"}

   907→
   908→      <p className="border-t border-teal/10 bg-white/40 py-1.5 text-center text-[10px] tracking-wide text-muted-ink/70">
