    95→    95→function splitSentences(text: string): string[] {
    96→    96→  const raw: string[] = [];
    97→    97→  let cur = "";
    98→    98→  for (const ch of text) {
    99→    99→    if (".!?؟۔".includes(ch) || ch === "\n") {
   100→   100→      const t = cur.trim();
   101→   101→      if (t) raw.push(t);
   102→   102→      cur = "";
   103→   103→    } else {
   104→   104→      cur += ch;
   105→   105→    }
   106→   106→  }
   107→   107→  if (cur.trim()) raw.push(cur.trim());
   108→   108→
   109→   109→  const merged: string[] = [];
   110→   110→  for (const part of raw) {
   111→   111→    const last = merged[merged.length - 1];
   112→   112→    if (last && (last.length < 14 || part.length < 14) && `${last} ${part}`.length <= 300) {
   113→   113→      merged[merged.length - 1] = `${last} ${part}`;
   114→   114→    } else {
   115→   115→      merged.push(part.slice(0, 320));
   116→   116→    }
   117→   117→  }
   118→   118→  return merged.slice(0, 8);
   119→   119→}
