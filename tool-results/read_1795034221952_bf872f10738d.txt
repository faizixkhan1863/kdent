    95→function splitSentences(text: string): string[] {
    96→  const raw: string[] = [];
    97→  let cur = "";
    98→  for (const ch of text) {
    99→    if (".!?؟۔".includes(ch) || ch === "\n") {
   100→      const t = cur.trim();
   101→      if (t) raw.push(t);
   102→      cur = "";
   103→    } else {
   104→      cur += ch;
   105→    }
   106→  }
   107→  if (cur.trim()) raw.push(cur.trim());
   108→
   109→  const merged: string[] = [];
   110→  for (const part of raw) {
   111→    const last = merged[merged.length - 1];
   112→    if (last && (last.length < 14 || part.length < 14) && `${last} ${part}`.length <= 300) {
   113→      merged[merged.length - 1] = `${last} ${part}`;
   114→    } else {
   115→      merged.push(part.slice(0, 320));
   116→    }
   117→  }
   118→  return merged.slice(0, 8);
   119→}

   125→async function fetchBlob(text: string, lang: Lang, emotion: Emotion): Promise<Blob> {
   126→  const ctrl = new AbortController();
   127→  const timer = setTimeout(() => ctrl.abort(), 20000);
   128→  try {
   129→    const res = await fetch("/api/voice-agent/tts", {
   130→      method: "POST",
   131→      headers: { "Content-Type": "application/json" },
   132→      body: JSON.stringify({ text, lang, emotion }),
   133→      signal: ctrl.signal,
   134→    });
   135→    if (!res.ok) throw new Error(`tts ${res.status}`);
   136→    const blob = await res.blob();
   137→    if (blob.size < 500) throw new Error("tiny audio");
   138→    return blob;
   139→  } finally {
   140→    clearTimeout(timer);
   141→  }
   142→}

   820→          }
   821→        }
